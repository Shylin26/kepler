    return 1 / (1 + math.exp(-x))

def sigmoid_derivative(x):
    return x * (1 - x)

def mse(predictions, targets):
    return sum((p - t) ** 2 for p, t in zip(predictions, targets)) / len(targets)

def train_neural_network(weights, biases, inputs, targets, epochs, initial_weights=0.0):
    if isinstance(initial_weights, float):
        weights = [[random.uniform(-initial_weights, initial_weights) for _ in range(len(inputs[0]))] for _ in range(len(biases))]

        errors = []
        
        for i in range(len(inputs)):

            a1 = [sigmoid(z) for z in z1]
